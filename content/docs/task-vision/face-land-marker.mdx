### Source Code
```tsx title="src/components/HandLandMark.tsx"
"use client";

import { useEffect, useRef, useState } from "react";
import { Camera, CameraOff, LoaderCircle } from "lucide-react";
import {
  DrawingUtils,
  FaceLandmarker,
  FilesetResolver,
} from "@mediapipe/tasks-vision";

type CameraPosition = "top-left" | "top-right" | "bottom-left" | "bottom-right";

interface FaceLandMarkerProps {
  position: CameraPosition;
  skeletonColor: string;
  dotColor: string;
  size: number;
}

const WASM_URL = "https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@1.1.0/wasm";
const MODEL_URL =
  "https://storage.googleapis.com/mediapipe-models/face_landmarker/face_landmarker/float16/1/face_landmarker.task";

function FaceLandMarker({
  position,
  skeletonColor,
  dotColor,
  size,
}: FaceLandMarkerProps) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const landmarkerRef = useRef<FaceLandmarker | null>(null);
  const streamRef = useRef<MediaStream | null>(null);
  const frameRef = useRef<number | null>(null);
  const runningRef = useRef(false);
  const startingRef = useRef(false);
  const [isModelReady, setIsModelReady] = useState(false);
  const [isCameraRunning, setIsCameraRunning] = useState(false);
  const [isCameraStarting, setIsCameraStarting] = useState(false);
  const [faceCount, setFaceCount] = useState(0);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const positionClasses: Record<CameraPosition, string> = {
    "top-left": "left-4 top-4 sm:left-6 sm:top-6",
    "top-right": "right-4 top-4 sm:right-6 sm:top-6",
    "bottom-left": "bottom-4 left-4 sm:bottom-6 sm:left-6",
    "bottom-right": "right-4 bottom-4 sm:right-6 sm:bottom-6",
  };

  useEffect(() => {
    let cancelled = false;

    async function createLandmarker() {
      try {
        const vision = await FilesetResolver.forVisionTasks(WASM_URL);
        const options = {
          baseOptions: { modelAssetPath: MODEL_URL },
          runningMode: "VIDEO" as const,
          numFaces: 2,
          minFaceDetectionConfidence: 0.5,
          minFacePresenceConfidence: 0.5,
          minTrackingConfidence: 0.5,
          outputFaceBlendshapes: false,
          outputFacialTransformationMatrixes: false,
        };

        let landmarker: FaceLandmarker;
        try {
          landmarker = await FaceLandmarker.createFromOptions(vision, {
            ...options,
            baseOptions: { ...options.baseOptions, delegate: "GPU" },
          });
        } catch (gpuError) {
          console.warn("GPU face landmarking unavailable; retrying with CPU:", gpuError);
          landmarker = await FaceLandmarker.createFromOptions(vision, {
            ...options,
            baseOptions: { ...options.baseOptions, delegate: "CPU" },
          });
        }

        if (cancelled) {
          landmarker.close();
          return;
        }
        landmarkerRef.current = landmarker;
        setIsModelReady(true);
      } catch (error) {
        console.error("Failed to load the face landmark model:", error);
        if (!cancelled) setErrorMessage("Could not load the face landmarker. Check your connection and reload.");
      }
    }

    void createLandmarker();
    return () => {
      cancelled = true;
      runningRef.current = false;
      if (frameRef.current !== null) window.cancelAnimationFrame(frameRef.current);
      streamRef.current?.getTracks().forEach((track) => track.stop());
      landmarkerRef.current?.close();
      landmarkerRef.current = null;
    };
  }, []);

  function stopCamera() {
    runningRef.current = false;
    setIsCameraRunning(false);
    setFaceCount(0);
    if (frameRef.current !== null) {
      window.cancelAnimationFrame(frameRef.current);
      frameRef.current = null;
    }
    streamRef.current?.getTracks().forEach((track) => track.stop());
    streamRef.current = null;
    if (videoRef.current) videoRef.current.srcObject = null;
    const canvas = canvasRef.current;
    const context = canvas?.getContext("2d");
    if (canvas && context) context.clearRect(0, 0, canvas.width, canvas.height);
  }

  async function toggleCamera() {
    if (runningRef.current) {
      stopCamera();
      return;
    }
    if (startingRef.current) return;

    startingRef.current = true;
    setIsCameraStarting(true);
    setErrorMessage(null);
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: false,
        video: { facingMode: "user" },
      });
      const video = videoRef.current;
      if (!video) {
        stream.getTracks().forEach((track) => track.stop());
        return;
      }
      streamRef.current = stream;
      video.srcObject = stream;
      await video.play();
      runningRef.current = true;
      setIsCameraRunning(true);
      frameRef.current = window.requestAnimationFrame(predictFrame);
    } catch (error) {
      streamRef.current?.getTracks().forEach((track) => track.stop());
      streamRef.current = null;
      console.error("Failed to start the camera:", error);
      setErrorMessage(
        error instanceof Error && error.name === "NotAllowedError"
          ? "Camera access was denied. Allow camera access in your browser settings and try again."
          : "Could not start the camera. Check that it is connected and not being used by another app.",
      );
    } finally {
      startingRef.current = false;
      setIsCameraStarting(false);
    }
  }

  function predictFrame(timestamp: number) {
    const video = videoRef.current;
    const canvas = canvasRef.current;
    const landmarker = landmarkerRef.current;
    if (!runningRef.current || !video || !canvas) return;

    if (!landmarker || video.readyState < HTMLMediaElement.HAVE_CURRENT_DATA || video.videoWidth === 0) {
      frameRef.current = window.requestAnimationFrame(predictFrame);
      return;
    }

    const context = canvas.getContext("2d");
    if (!context) {
      setErrorMessage("Could not create the face landmark canvas.");
      stopCamera();
      return;
    }

    if (canvas.width !== video.videoWidth || canvas.height !== video.videoHeight) {
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
    }

    try {
      const result = landmarker.detectForVideo(video, timestamp);
      context.clearRect(0, 0, canvas.width, canvas.height);
      setFaceCount(result.faceLandmarks.length);
      const drawing = new DrawingUtils(context);
      for (const landmarks of result.faceLandmarks) {
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_TESSELATION, {
          color: `${skeletonColor}55`,
          lineWidth: Math.max(0.5, size * 0.7),
        });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_FACE_OVAL, {
          color: skeletonColor,
          lineWidth: Math.max(1, size * 1.5),
        });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_LIPS, {
          color: skeletonColor,
          lineWidth: Math.max(1, size * 1.5),
        });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_LEFT_EYE, { color: dotColor, lineWidth: size });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_RIGHT_EYE, { color: dotColor, lineWidth: size });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_LEFT_EYEBROW, { color: dotColor, lineWidth: size });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_RIGHT_EYEBROW, { color: dotColor, lineWidth: size });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_LEFT_IRIS, { color: dotColor, lineWidth: size });
        drawing.drawConnectors(landmarks, FaceLandmarker.FACE_LANDMARKS_RIGHT_IRIS, { color: dotColor, lineWidth: size });
      }
    } catch (error) {
      console.error("Face landmarking failed:", error);
      setErrorMessage("Face landmarking failed. Please restart the camera.");
      stopCamera();
      return;
    }

    if (runningRef.current) frameRef.current = window.requestAnimationFrame(predictFrame);
  }

  return (
    <main className="relative min-h-screen w-full overflow-hidden bg-[#090a0c]">
      <div className={`absolute z-10 aspect-video w-[min(24rem,calc(100vw-2rem))] overflow-hidden rounded-2xl border border-white/[0.09] bg-[#0e1012] shadow-2xl shadow-black/50 sm:w-[min(28rem,calc(100vw-3rem))] ${positionClasses[position]}`}>
        <video ref={videoRef} autoPlay muted playsInline aria-label="Camera preview" className={`absolute inset-0 size-full scale-x-[-1] object-cover ${isCameraRunning ? "" : "hidden"}`} />
        <canvas ref={canvasRef} aria-hidden="true" className={`absolute inset-0 size-full scale-x-[-1] object-cover ${isCameraRunning ? "block" : "hidden"}`} />
        {!isCameraRunning && (
          <div className="absolute inset-0 flex flex-col items-center justify-center px-4 text-center">
            <CameraOff className="mb-3 size-7 text-sky-300" aria-hidden="true" />
            <p className="text-sm font-medium text-[#f2f3f4]">{errorMessage ?? "Camera is off"}</p>
            <p className="mt-1 text-xs text-[#92969b]">{isModelReady ? "Press the camera button to start" : errorMessage ? "Face landmarking could not be initialized" : "Face landmarker is loading"}</p>
          </div>
        )}
        {isCameraRunning && (
          <div className="absolute bottom-3 left-3 rounded-full border border-white/10 bg-[#101311]/90 px-2.5 py-1 text-xs text-[#f2f3f4] shadow-lg backdrop-blur" role="status" aria-live="polite">
            {faceCount === 0 ? "No face detected" : `${faceCount} ${faceCount === 1 ? "face" : "faces"} tracked`}
          </div>
        )}
        <button type="button" onClick={() => void toggleCamera()} disabled={!isModelReady || isCameraStarting} aria-label={isCameraStarting ? "Starting camera" : isCameraRunning ? "Turn camera off" : "Turn camera on"} title={isCameraStarting ? "Starting camera" : isCameraRunning ? "Turn camera off" : "Turn camera on"} className="absolute right-3 top-3 z-20 flex size-10 items-center justify-center rounded-full border border-white/10 bg-[#101311]/90 text-white shadow-lg backdrop-blur transition hover:border-sky-300/40 hover:bg-[#171c19] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-sky-300 disabled:cursor-wait disabled:opacity-70">
          {isCameraStarting ? <LoaderCircle className="size-4 animate-spin text-sky-300" aria-hidden="true" /> : isCameraRunning ? <CameraOff className="size-4 text-[#d1d5d8]" aria-hidden="true" /> : <Camera className="size-4 text-sky-300" aria-hidden="true" />}
        </button>
      </div>
    </main>
  );
}

export default FaceLandMarker;
```
